import importlib.util
import random
import sys
from collections import deque
from datetime import timedelta
from pathlib import Path
from uuid import UUID

from tradeagent.scalping_market import datetime_ns, decode_crypto_message
from tradeagent.scalping_research import _QuoteReplay
from tradeagent.shadow_dataset import WINDOW_START, ShadowDatasetCollector, ShadowDatasetProtocol

path = Path(__file__).with_name("pre_v2_feasibility_study.py")
spec = importlib.util.spec_from_file_location("feasibility", path)
module = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = module
spec.loader.exec_module(module)
checks = 0


def trace(messages):
    global checks
    engine = module.QuoteKernel()
    reference = _QuoteReplay(("BTC/USD", "ETH/USD"), 2000)
    for index, (message, provider, received) in enumerate(messages, 1):
        row = {"S": "BTC/USD", "t": provider.isoformat(), **message}
        event = decode_crypto_message(
            row, connection_id=UUID(int=1), receive_sequence=index,
            received_at=received, received_monotonic_ns=index,
        )
        actual = engine.on_message(
            row, datetime_ns(received), datetime_ns(received + timedelta(microseconds=10)),
            f"synthetic:{index}", "us-1",
        )
        expected, _ = reference.on_event(event)
        assert (actual is None) == (expected is None), (row, actual, expected)
        if expected:
            assert actual["bid"] == str(expected.bid)
            assert actual["ask"] == str(expected.ask)
            assert actual["bid_size"] == str(expected.bid_size)
            assert actual["ask_size"] == str(expected.ask_size)
            assert actual["provider_ns"] == expected.exchange_time_ns
            assert actual["received_ns"] == datetime_ns(expected.received_at)
            assert actual["location"] == "us-1"
        checks += 1


quote = {"T": "q", "bp": "100", "ap": "100.01", "bs": "2", "as": "2"}
snapshot = {
    "T": "o", "r": True, "b": [{"p": "100", "s": "2"}],
    "a": [{"p": "100.01", "s": "2"}],
}
delta = {"T": "o", "b": [{"p": "100", "s": "3"}], "a": []}
at = WINDOW_START
trace([
    (quote, at, at + timedelta(milliseconds=10)),
    (delta, at + timedelta(milliseconds=100), at + timedelta(milliseconds=110)),
    (snapshot, at - timedelta(seconds=10), at + timedelta(milliseconds=120)),
    (delta, at + timedelta(milliseconds=200), at + timedelta(milliseconds=210)),
    (quote, at + timedelta(milliseconds=300), at + timedelta(milliseconds=310)),
])
trace([
    (snapshot, at, at + timedelta(milliseconds=10)),
    ({"T": "o", "a": [{"p": "99", "s": "1"}]}, at + timedelta(seconds=1),
     at + timedelta(seconds=1.01)),
    (quote, at + timedelta(seconds=.5), at + timedelta(seconds=1.1)),
])
trace([
    (snapshot, at, at + timedelta(milliseconds=10)),
    ({**quote, "bs": "0"}, at + timedelta(seconds=1), at + timedelta(seconds=1.01)),
    (delta, at + timedelta(seconds=2), at + timedelta(seconds=2.01)),
])
rng = random.Random(20261002)
for _ in range(100):
    seq = [(snapshot, at, at + timedelta(milliseconds=10))]
    elapsed = 0
    for index in range(20):
        elapsed += rng.choice([.1, .5, 1, 3, 7])
        provider = at + timedelta(seconds=elapsed)
        received = provider + timedelta(milliseconds=rng.choice([10, 30, 150, 2500]))
        message = rng.choice([quote, delta, snapshot])
        seq.append((message, provider, received))
    trace(seq)

p = ShadowDatasetProtocol(
    frozen_at=WINDOW_START - timedelta(hours=1), code_sha="a" * 40,
    account_digest="b" * 64,
)
for entry_delay, future_delay, entry_size, future_size in (
    (.5, .5, "2", "2"), (3, .5, "2", "2"), (.5, 3, "2", "2"),
    (.5, .5, ".01", "2"), (.5, .5, "2", ".01"),
):
    c = ShadowDatasetCollector(p)
    when = WINDOW_START + timedelta(seconds=5)
    q0 = decode_crypto_message(
        {"S": "BTC/USD", "t": (when - timedelta(seconds=entry_delay)).isoformat(),
         **quote, "as": entry_size},
        connection_id=UUID(int=1), receive_sequence=1,
        received_at=when - timedelta(seconds=entry_delay) + timedelta(milliseconds=10),
        received_monotonic_ns=1,
    )
    c.on_market(q0, q0.received_at)
    evaluation = c.evaluate(when)[0]["values"]["payload"]
    q1 = decode_crypto_message(
        {"S": "BTC/USD", "t": (when + timedelta(seconds=60-future_delay)).isoformat(),
         **quote, "bs": future_size},
        connection_id=UUID(int=1), receive_sequence=2,
        received_at=when + timedelta(seconds=60-future_delay) + timedelta(milliseconds=10),
        received_monotonic_ns=2,
    )
    c.on_market(q1, q1.received_at)
    labels = c.resolve(when + timedelta(seconds=62.1))
    label = next(r["values"]["payload"] for r in labels if (
        r["values"]["evaluation_id"] == evaluation["evaluation_id"]
        and r["values"]["horizon_seconds"] == 60
    ))
    reasons = module.measure(
        module.normalized(evaluation["decision_observation"]),
        module.normalized(label["exit_observation"]),
        datetime_ns(when), datetime_ns(when + timedelta(seconds=60)),
    )
    assert reasons == label["missing_reasons"]
    checks += 1

q = {
    "provider_ns": datetime_ns(at), "received_ns": datetime_ns(at),
    "processed_ns": datetime_ns(at), "bid": "100", "ask": "100.01",
    "bid_size": "2", "ask_size": "2", "epoch": 1,
}
later = {**q, "received_ns": datetime_ns(at + timedelta(seconds=1)),
         "processed_ns": datetime_ns(at + timedelta(seconds=1))}
assert module.select_at(deque([q, later]), datetime_ns(at), datetime_ns(at + timedelta(seconds=2)), False) == q
assert module.select_at(deque([later]), datetime_ns(at), datetime_ns(at), True) is None
checks += 2
print(f"{checks} source-kernel and exact primary-contract equivalence checks passed")
