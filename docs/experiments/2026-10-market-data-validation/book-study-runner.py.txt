"""One separately versioned, public-only checksum book feasibility study."""

import asyncio
import base64
import importlib.util
import json
import signal
import sys
import tarfile
import time
from datetime import UTC, datetime
from hashlib import sha256
from pathlib import Path

import httpx
from websockets.asyncio.client import connect
from websockets.exceptions import ConnectionClosed, InvalidHandshake
from kraken_checksum_book import Book
from tradeagent.config import AppConfig
from tradeagent.persistence import Database
from tradeagent.scalping_market import datetime_ns, timestamp_ns
from tradeagent.scalping_store import canonical
from tradeagent.shadow_dataset import ShadowDatasetProtocol

ROOT = Path(__file__).parent
spec = importlib.util.spec_from_file_location("book_study_core", ROOT/"pre_v2_feasibility_study.py")
core = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = core
spec.loader.exec_module(core)
LOCATION = "kraken-spot-checksum-book10"


class Capture(core.Capture):
    def __init__(self, output):
        super().__init__(LOCATION, output)
        self.books = {symbol: Book() for symbol in core.SYMBOLS}
        self.epoch = 0
        self.acknowledgments = []
        self.last_received = None

    def invalidate(self, reason):
        for book in self.books.values():
            book.invalidate(reason)
        self.epoch += 1
        self.counts["book_invalidations"] += 1
        self.state("book_invalidated", reason=reason)

    async def run(self, stop, genesis):
        self.root = genesis
        attempts = 0
        try:
            while not stop.is_set() and attempts < 3:
                attempts += 1
                self.connection = str(core.uuid4())
                self.sequence = 0
                self.invalidate("new_connection_requires_snapshot")
                self.state("connecting_public")
                try:
                    async with connect("wss://ws.kraken.com/v2", ping_interval=20,
                                       ping_timeout=20, close_timeout=5, max_queue=16,
                                       max_size=2*1024**2) as ws:
                        self.connected = True
                        self.state("connected_public")
                        await ws.send(json.dumps({"method":"subscribe","params":{
                            "channel":"book","symbol":list(core.SYMBOLS),
                            "depth":10,"snapshot":True},"req_id":1}))
                        while not stop.is_set():
                            raw = await ws.recv()
                            received = datetime_ns(datetime.now(UTC))
                            monotonic = time.monotonic_ns()
                            if isinstance(raw,bytes):
                                raw = raw.decode("utf-8")
                            if self.last_received is not None and received < self.last_received:
                                raise ValueError("local_receive_clock_regression")
                            self.last_received = received
                            msg = json.loads(raw,parse_float=core.Decimal)
                            self.frame_index += 1
                            envelope = {
                                "location":LOCATION,"endpoint":"wss://ws.kraken.com/v2",
                                "channel":"book","connection_id":self.connection,
                                "frame_index":self.frame_index,"received_ns":received,
                                "received_at":core.iso_ns(received),
                                "received_monotonic_ns":monotonic,"payload_text":raw,
                                "journal_append_started_at":datetime.now(UTC).isoformat(),
                            }
                            self.root=sha256((self.root+canonical(envelope)).encode()).hexdigest()
                            self.raw.write(canonical({"root":self.root,**envelope})+"\n")
                            self.counts["frames"]+=1
                            if self.frame_index%100==0:
                                self.raw.flush()
                                if self.raw_path.stat().st_size>core.RAW_LIMIT:
                                    raise ValueError("raw_capture_budget_exceeded")
                            if msg.get("method")=="subscribe":
                                self.acknowledgments.append(msg)
                                if msg.get("success") is not True:
                                    raise ValueError("book_subscription_rejected")
                                self.state("subscribed",symbol=msg["result"]["symbol"])
                            elif msg.get("channel")=="book":
                                for row in msg["data"]:
                                    self.sequence+=1
                                    if row["symbol"] not in self.books:
                                        raise ValueError("book_symbol_mismatch")
                                    provider=timestamp_ns(row["timestamp"])
                                    if provider>received:
                                        raise ValueError("provider_timestamp_after_receipt")
                                    processed=datetime_ns(datetime.now(UTC))
                                    book=self.books[row["symbol"]]
                                    quote=book.apply(row,msg["type"],provider)
                                    self.counts["book_messages"]+=1
                                    if quote is None:
                                        self.counts["rejected_book_messages"]+=1
                                        if book.last_reason=="update_before_valid_snapshot":
                                            continue
                                        reason=book.last_reason
                                        self.invalidate(reason)
                                        raise ValueError(reason)
                                    accepted=datetime_ns(datetime.now(UTC))
                                    obs={
                                        **quote,"event_id":f"{LOCATION}:{self.connection}:{self.sequence}",
                                        "symbol":row["symbol"],"location":LOCATION,
                                        "source_kind":"checksum_validated_native_book",
                                        "provider_timestamp":row["timestamp"],"provider_ns":provider,
                                        "received_ns":received,"received_at":core.iso_ns(received),
                                        "processing_started_ns":processed,
                                        "processed_ns":accepted,"processed_at":core.iso_ns(accepted),
                                        "accepted_ns":accepted,"epoch":self.epoch,
                                        "connection_id":self.connection,
                                        "raw_frame_index":self.frame_index,
                                        "journal_append_started_at":datetime.now(UTC).isoformat(),
                                    }
                                    self.quotes.write(canonical(obs)+"\n")
                                    self.counts["checksum_valid_quotes"]+=1
                            elif msg.get("channel") not in ("heartbeat","status") and msg.get("method")!="pong":
                                raise ValueError("unexpected_book_frame")
                except (ConnectionClosed,InvalidHandshake,OSError,TimeoutError,ValueError,KeyError) as error:
                    self.connected=False
                    self.failure=str(error) if isinstance(error,(ValueError,KeyError)) else type(error).__name__
                    self.invalidate(self.failure)
                    self.state("connection_failure",error=self.failure)
                    if "budget" in self.failure or "subscription" in self.failure:
                        break
                    if not stop.is_set():
                        self.counts["reconnects"]+=1
                        await asyncio.sleep(5)
        finally:
            self.connected=False
            self.finished=True
            self.state("planned_close" if stop.is_set() else "capture_failed")
            for book in self.books.values():
                book.invalidate("closed_connection")
            self.raw.close()
            self.quotes.close()


async def main(plan_path,output):
    signal.alarm(3900)
    plan=json.loads(plan_path.read_text())
    digest=plan.pop("protocol_hash")
    if sha256(canonical(plan).encode()).hexdigest()!=digest:
        raise ValueError("book study freeze hash mismatch")
    plan["protocol_hash"]=digest
    if datetime.now(UTC)>=datetime.fromisoformat(plan["evaluation_start"]):
        raise ValueError("missed predeclared book study start")
    output.mkdir()
    (output/"frozen-study.json").write_text(json.dumps(plan,indent=2,sort_keys=True)+"\n")
    with Database(AppConfig().database_url.get_secret_value(),pool_size=1) as db:
        with db.begin() as conn:
            conn.exec_driver_sql("SET TRANSACTION READ ONLY")
            protocol=ShadowDatasetProtocol.model_validate(conn.scalar(
                core.select(core.shadow_datasets.c.protocol).where(
                    core.shadow_datasets.c.dataset_id==core.DATASET_ID)))
        with httpx.Client(timeout=10,follow_redirects=False) as client:
            before=core.guard(db,protocol,client)
    capture=Capture(output);stop=asyncio.Event()
    task=asyncio.create_task(capture.run(stop,digest))
    class Inert:
        def __init__(self,location,output):
            self.location,self.health=location,[]
        async def run(self,settings,stopped,genesis):
            await stopped.wait()
    old=core.Capture;core.Capture=Inert
    try:
        _,interruption=await core.collect(plan,output,protocol)
    finally:
        core.Capture=old;stop.set();task.cancel()
        try:
            await task
        except asyncio.CancelledError:
            pass
    (output/f"{LOCATION}-connection-health.json").write_text(json.dumps(capture.health,indent=2)+"\n")
    (output/"subscription-acks.json").write_text(canonical(capture.acknowledgments)+"\n")
    with Database(AppConfig().database_url.get_secret_value(),pool_size=1) as db:
        anchors,manifest=core.load_anchors(db,plan,output)
        with httpx.Client(timeout=10,follow_redirects=False) as client:
            after=core.guard(db,protocol,client)
    coverage=core.evaluate_source(LOCATION,anchors,output,capture)
    passed=all(r["matured_evaluations"]==180 and r["coverage"]>=.95 for r in coverage)
    passed=bool(passed and not interruption and capture.failure is None and not capture.counts["reconnects"])
    summary={
        "schema":"native-checksum-book-feasibility-v1","study_protocol":plan,
        "completed_at":datetime.now(UTC).isoformat(),"coverage":coverage,
        "data_quality_gate_passed":passed,"overall_v2_ready":False,
        "source_migration_or_configuration_changed":False,"v2_started":False,
        "clock_semantics":"provider timestamp of checksum-validated book state/update, not last BBO price change",
        "old_ticker_or_v1_evidence_changed":False,"checksum_guarantees":"local top10 reconstruction consistency only",
        "account_specific_execution_eligibility_proven":False,
        "public_only":True,"orders_submitted":0,"private_kraken_requests":0,
        "capture_failure":capture.failure,"capture_counts":dict(capture.counts),
        "raw_manifest_root":capture.root,"v1_anchor_manifest":manifest,
        "broker_before":before,"broker_after":after,"feasibility_interruption":interruption,
        "artifacts":[],
    }
    for file in sorted(output.iterdir()):
        data=file.read_bytes()
        summary["artifacts"].append({"path":file.name,"bytes":len(data),"sha256":sha256(data).hexdigest()})
    (output/"readiness.json").write_text(json.dumps(summary,indent=2,sort_keys=True)+"\n")
    archive=output.parent/"feasibility-bundle.tar.gz"
    with tarfile.open(archive,"w:gz") as bundle:
        for file in sorted(output.iterdir()):
            bundle.add(file,arcname=file.name,recursive=False)
    print("BOOK_STUDY_RESULT="+json.dumps({k:v for k,v in summary.items() if k!="artifacts"},sort_keys=True),flush=True)
    data=archive.read_bytes();encoded=base64.b64encode(data).decode()
    chunks=[encoded[i:i+6000]for i in range(0,len(encoded),6000)]
    for index,chunk in enumerate(chunks):
        print("FEASIBILITY_BLOB="+json.dumps({"name":archive.name,"index":index,"chunks":len(chunks),
            "bytes":len(data),"sha256":sha256(data).hexdigest(),"data":chunk},separators=(",",":")),flush=True)
        time.sleep(.025)
    print("FEASIBILITY_FINISHED=true",flush=True)


if __name__=="__main__":
    asyncio.run(main(Path(sys.argv[1]),Path(sys.argv[2])))
