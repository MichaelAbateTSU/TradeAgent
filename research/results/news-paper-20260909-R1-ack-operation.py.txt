def require(condition, reason):
    if not condition:
        raise RuntimeError("NEWS_ACK_ABORTED: " + reason)


def digest(value):
    return hashlib.sha256(
        json.dumps(value, sort_keys=True, default=str, separators=(",", ":")).encode()
    ).hexdigest()


def control_snapshot(connection):
    return {
        row.control_key: (row.control_value, row.updated_at)
        for row in connection.execute(select(controls).order_by(controls.c.control_key))
    }


def validate_controls(state):
    require(state.get("kill_switch") == ("active", GLOBAL_AT),
            "global kill value/original timestamp differs")
    require(state.get(PAUSE) == (R1, PAUSE_AT), "reviewed pause value/timestamp differs")
    require(all(COHORT + ":" + suffix not in state for suffix in AUTHORITY_KEYS),
            "cohort authority/certificate/terminal already exists")
    budget = state.get(session_control_key(ACCOUNT, date(2026, 9, 9)))
    if budget:
        value = json.loads(budget[0])
        require(
            value["account_digest"] == ACCOUNT
            and value["planned_session_date"] == "2026-09-09"
            and value["total_entries_reserved"] == 0
            and value["news_entries_reserved"] == 0
            and value["equipment_client_order_id"] is None,
            "account/session already reserved or attempted",
        )


def validate_evidence(connection):
    versions = {
        row.evidence_id: row.payload for row in connection.execute(
            select(event_evidence).where(event_evidence.c.payload["source_url"].as_string() == URL)
        )
    }
    require(set(versions) == APPROVED_IDS, "new or missing source lineage version")
    for frozen in EXPECTED["versions"]:
        payload = versions[frozen["evidence_id"]]
        for key in (
            "source_url", "source_version", "headline", "content_sha256", "revision_of",
            "published_at", "provider_created_at", "provider_updated_at", "first_received_at",
            "content_available_at", "is_primary_source", "is_correction", "is_retraction",
            "raw_payload_sha256",
        ):
            require(payload.get(key) == frozen.get(key), "source field changed: " + key)
        require(payload["content_sha256"] == CONTENT
                and hashlib.sha256(payload["content"].encode()).hexdigest() == CONTENT,
                "source content hash changed")
        raw = json.loads(payload["raw_metadata_json"])["raw_document"]
        require(hashlib.sha256(raw.encode()).hexdigest() == frozen["raw_document_sha256"],
                "retained raw document changed")
        require(payload["published_at"] is None and not payload["is_correction"]
                and not payload["is_retraction"], "publication or explicit correction changed")
    decisions = [dict(row) for row in connection.execute(
        select(event_decisions, event_candidate_states.c.status.label("candidate_status"))
        .outerjoin(event_candidate_states,
                   event_candidate_states.c.decision_id == event_decisions.c.decision_id)
        .where(event_decisions.c.cohort_id == COHORT)
    ).mappings()]
    reviews = [row for row in decisions if row["payload"].get("position_review_required")]
    require(len(reviews) == 3 and {row["evidence_id"] for row in reviews} == APPROVED_IDS,
            "new or missing position-review decision")
    projections = []
    for row in reviews:
        frozen = next(item for item in EXPECTED["all_position_review_decisions"]
                      if item["evidence_id"] == row["evidence_id"])
        require(row["decision_id"] == frozen["decision_id"]
                and str(row["decided_at"]) == frozen["decided_at"]
                and row["candidate_status"] == frozen["candidate_status"] == "rejected",
                "decision identity/time/candidate state changed")
        payload = row["payload"]
        for key in ("action", "hypothesis", "reasons", "position_review_required"):
            require(payload.get(key) == frozen.get(key), "decision changed: " + key)
        extraction = payload["extraction"]
        require(extraction["facts"] == frozen["facts"] == []
                and extraction["event_type"] == frozen["event_type"] == "unclassified"
                and payload["hypothesis"] is None and payload["action"] == "abstain",
                "economic extraction differs")
        projections.append(frozen)
    require(connection.scalar(select(event_candidate_states.c.decision_id).where(
        event_candidate_states.c.status == "waiting",
    ).limit(1)) is None, "waiting local news candidate exists")
    decided = select(event_decisions.c.evidence_id).where(event_decisions.c.cohort_id == COHORT)
    require(connection.scalar(select(event_evidence.c.evidence_id).where(
        event_evidence.c.evidence_id.not_in(decided),
        event_evidence.c.payload["is_primary_source"].as_boolean().is_(True),
        (event_evidence.c.payload["is_correction"].as_boolean().is_(True)
         | event_evidence.c.payload["is_retraction"].as_boolean().is_(True)
         | event_evidence.c.payload["revision_of"].as_string().is_not(None)),
    ).limit(1)) is None, "pending unreviewed primary revision/risk")
    pauses = [dict(row) for row in connection.execute(select(events).where(
        events.c.trace_id == COHORT, events.c.event_type == "event_pause",
        events.c.payload["reason"].as_string() == R1,
    ).order_by(events.c.occurred_at)).mappings()]
    require(len(pauses) == 3 and {row["event_id"] for row in pauses} == APPROVED_PAUSE_IDS,
            "R1 event set changed")
    for row in pauses:
        frozen = next(item for item in EXPECTED["all_cohort_R1_pause_events"]
                      if item["event_id"] == row["event_id"])
        require(row["payload"] == frozen["payload"] == {"reason": R1}
                and str(row["occurred_at"]) == frozen["occurred_at"],
                "R1 event contents/time changed")
    latest = connection.execute(select(events).where(
        events.c.trace_id == COHORT, events.c.event_type == "event_pause",
    ).order_by(events.c.occurred_at.desc()).limit(1)).mappings().one()
    require(latest["event_id"] == pauses[-1]["event_id"], "newer unrelated risk pause")
    return {
        "source_payloads_sha256": digest(versions),
        "review_decisions_sha256": digest(projections),
        "R1_pause_event_ids": [row["event_id"] for row in pauses],
    }


require(os.environ.get("RENDER_GIT_COMMIT") == CODE, "deployed image mismatch")
require(EXPECTED["cohort_id"] == COHORT and EXPECTED["cohort"]["config_hash"] == CONFIG
        and EXPECTED["code_sha"] == CODE, "frozen review identity differs")
require({row["evidence_id"] for row in EXPECTED["versions"]} == APPROVED_IDS
        and len(EXPECTED["versions"]) == 3, "unapproved evidence scope")
with Database(AppConfig().database_url.get_secret_value()) as database:
    with database.begin() as connection:
        baseline = control_snapshot(connection)
        validate_controls(baseline)
    with httpx.Client(timeout=2) as client:
        broker = AlpacaPaperClient(AlpacaPaperSettings.model_validate({}), client=client)
        with database.begin() as connection:
            connection.execute(text("SET LOCAL lock_timeout = '1s'"))
            connection.execute(text("SET LOCAL statement_timeout = '8s'"))
            connection.execute(text("SET LOCAL idle_in_transaction_session_timeout = '12s'"))
            connection.execute(text(
                "LOCK TABLE controls_v2, events_v2, event_evidence, event_decisions, "
                "event_candidate_states, event_order_links, orders, event_cohorts "
                "IN SHARE ROW EXCLUSIVE MODE NOWAIT"
            ))
            locked_at = time.monotonic()
            current = control_snapshot(connection)
            require(current == baseline, "a control changed before acquiring locks")
            connection.execute(select(controls).where(
                controls.c.control_key.in_([PAUSE, "kill_switch"])
            ).with_for_update()).all()
            cohort = connection.execute(select(event_cohorts).where(
                event_cohorts.c.cohort_id == COHORT
            ).with_for_update()).mappings().one()
            require(cohort["config_hash"] == CONFIG and cohort["manifest"]["code_sha"] == CODE
                    and cohort["manifest"]["settings"]["entry_policy"] == "news-paper",
                    "immutable cohort/config/policy changed")
            require(connection.scalar(select(events.c.event_id).where(
                events.c.event_id == AUDIT_ID)) is None, "acknowledgement already exists")
            reviewed_state = validate_evidence(connection)
            require(connection.scalar(select(orders.c.order_id).where(
                orders.c.status.not_in(FINAL)).limit(1)) is None, "unresolved local order")
            require(connection.scalar(select(event_order_links.c.client_order_id).where(
                event_order_links.c.cohort_id == COHORT).limit(1)) is None,
                "cohort already has order history")
            account = broker.account()
            positions = broker.positions()
            open_orders = broker.open_orders()
            clock = broker.clock()
            broker_at = datetime.now(UTC)
            require(hashlib.sha256(str(account.id).encode()).hexdigest() == ACCOUNT,
                    "actual broker account differs")
            require(account.status == "ACTIVE" and not account.account_blocked
                    and not account.trading_blocked and not positions and not open_orders,
                    "broker not ACTIVE/unblocked/flat/unreserved")
            require(not clock.is_open, "ack restricted to this paused after-hours review")
            require(datetime(2026, 9, 9, 5, 28, 35, 403000, tzinfo=UTC)
                    <= clock.timestamp.astimezone(UTC)
                    < datetime(2026, 9, 9, 6, 30, tzinfo=UTC),
                    "one-use operator acknowledgement time window expired")
            require(time.monotonic() - locked_at < 5, "fresh review exceeded time bound")
            now = connection.scalar(text("SELECT clock_timestamp()"))
            payload = {
                "review_id": AUDIT_ID, "cohort_id": COHORT, "code_sha": CODE,
                "config_hash": CONFIG, "account_digest": ACCOUNT,
                "planned_session_date": "2026-09-09", "reviewed_at": now.isoformat(),
                "parent_approved_at": "2026-09-09T01:28:35.403-04:00",
                "operation_expiry": "2026-09-09T06:30:00Z",
                "authority": "NEW separate parent-approved exact-evidence acknowledgement",
                "prior_cohort_acknowledgement_inherited": False,
                "evidence_ids": sorted(APPROVED_IDS), "content_sha256": CONTENT,
                "source_versions": [
                    {key: item[key] for key in (
                        "evidence_id", "source_version", "raw_document_sha256", "revision_of"
                    )} for item in EXPECTED["versions"]
                ],
                "source_url": URL,
                "basis": {
                    "classification": "Three unchanged metadata/cache-only Pusheen source versions",
                    "economic_facts": [], "event_type": "unclassified", "hypothesis": None,
                    "candidates": "all rejected", "publication": "unknown; unchanged",
                    "explicit_correction_or_retraction": False,
                    "source_and_raw_hashes_match_fresh_review": True,
                    "no_position_or_queued_news_context_affected": True,
                },
                "evidence_artifact_sha256": EXPECTED["artifact_sha256"],
                "locked_frozen_state": reviewed_state,
                "broker_checked_at": broker_at.isoformat(),
                "broker_clock": clock.model_dump(mode="json"), "broker_status": account.status,
                "broker_positions": 0, "broker_open_orders": 0,
                "unresolved_local_intents": 0, "waiting_local_news_candidates": 0,
                "controls_before_sha256": digest(baseline),
                "pause_before": [baseline[PAUSE][0], baseline[PAUSE][1].isoformat()],
                "pause_after": "OPERATOR_PAUSE",
                "global_kill_unchanged": [
                    baseline["kill_switch"][0], baseline["kill_switch"][1].isoformat()
                ],
                "news_authorization_issued": False, "certificate_issued": False,
                "orders_submitted": 0, "future_R1_requires_separate_fresh_review": True,
                "publication_and_decisions_unchanged": True,
                "incident_acceptance_and_explicit_news_approval_still_required": True,
            }
            changed = connection.execute(update(controls).where(
                controls.c.control_key == PAUSE,
                controls.c.control_value == baseline[PAUSE][0],
                controls.c.updated_at == baseline[PAUSE][1],
            ).values(control_value="OPERATOR_PAUSE", updated_at=now))
            require(changed.rowcount == 1, "pause compare-and-set failed")
            connection.execute(insert(events).values(
                event_id=AUDIT_ID, occurred_at=now, recorded_at=now,
                event_type="event_operator_risk_review", trace_id=COHORT, payload=payload,
            ))
            append_reporting_metadata(connection, AUDIT_ID, "event_operator_risk_review", payload)
            transaction_after = control_snapshot(connection)
            require({key: value for key, value in transaction_after.items() if key != PAUSE}
                    == {key: value for key, value in baseline.items() if key != PAUSE},
                    "another control changed inside transaction")
            require(connection.scalar(select(event_reporting_metadata.c.event_id).where(
                event_reporting_metadata.c.event_id == AUDIT_ID)) == AUDIT_ID,
                "atomic reporting metadata missing")
            require(time.monotonic() - locked_at < 5, "atomic review exceeded time bound")
    with database.begin() as connection:
        stored = dict(connection.execute(select(events).where(
            events.c.event_id == AUDIT_ID)).mappings().one())
        after = control_snapshot(connection)
        sidecar = dict(connection.execute(select(event_reporting_metadata).where(
            event_reporting_metadata.c.event_id == AUDIT_ID)).mappings().one())
    print("NEWS_R1_ACK " + json.dumps({
        "committed": True, "audit_event": stored, "reporting_metadata": sidecar,
        "post_commit_verified_at": datetime.now(UTC).isoformat(),
        "pause": after[PAUSE], "global_kill": after["kill_switch"],
        "post_commit_pause_matches": after[PAUSE] == ("OPERATOR_PAUSE", now),
        "post_commit_global_unchanged": after["kill_switch"] == baseline["kill_switch"],
        "all_other_controls_unchanged_in_transaction": True,
        "post_commit_other_control_changes": [
            key for key in set(after) | set(baseline)
            if key != PAUSE and after.get(key) != baseline.get(key)
        ],
        "cohort_authority_or_certificate_exists": any(
            COHORT + ":" + suffix in after for suffix in AUTHORITY_KEYS
        ),
        "orders_submitted": 0,
    }, default=str, sort_keys=True), flush=True)
